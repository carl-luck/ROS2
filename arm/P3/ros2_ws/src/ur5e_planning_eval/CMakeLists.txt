cmake_minimum_required(VERSION 3.8)
project(ur5e_planning_eval)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
add_executable(find_detour_case src/find_detour_case.cpp)
ament_target_dependencies(find_detour_case rclcpp rcl_interfaces moveit_core moveit_ros_planning moveit_msgs)
add_executable(check_spline_samples src/check_spline_samples.cpp)
ament_target_dependencies(check_spline_samples rclcpp rcl_interfaces moveit_core moveit_ros_planning moveit_msgs)
install(TARGETS find_detour_case check_spline_samples DESTINATION lib/${PROJECT_NAME})
ament_package()
